if(CONFIG_RTBUS_LIB)
  zephyr_linker_sources(SECTIONS rtbus_sections.ld)

  target_sources(app PRIVATE
    src/rtbus.c
  )
  if(CONFIG_RTBUS_ITERABLE_ENTRY)
    target_sources(app PRIVATE src/rtbus_iterable.c)
  else()
    target_sources(app PRIVATE src/rtbus_entries.c)
  endif()

  target_sources_ifdef(CONFIG_RTBUS_NATIVE_SERVICE app PRIVATE
    services/rtbus_runtime.c
    services/rtbus_diagnostics.c
    services/rtbus_cli.c
    services/rtbus_ymodem.c
    services/rtbus_event.c
    services/native_service/native_service.c
    services/native_service/native_board.c
    services/rtbus_image.c
  )

  target_include_directories(app PRIVATE
    include
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../system/framework/abi/include
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../system/framework/include
  )

  if(CONFIG_RTBUS_IMAGE_VERIFY_P256)
    if(NOT RTBUS_IMAGE_PUBLIC_KEY_FILE OR NOT EXISTS "${RTBUS_IMAGE_PUBLIC_KEY_FILE}")
      message(FATAL_ERROR "P-256 verification requires RTBUS_IMAGE_PUBLIC_KEY_FILE (65-byte SEC1 public key)")
    endif()
    file(SIZE "${RTBUS_IMAGE_PUBLIC_KEY_FILE}" public_key_size)
    if(NOT public_key_size EQUAL 65)
      message(FATAL_ERROR
        "P-256 public key ${RTBUS_IMAGE_PUBLIC_KEY_FILE} is ${public_key_size} bytes; expected 65 binary SEC1 bytes. Do not edit this file as text. Use the packer's --public-key command to derive it from the private key.")
    endif()
    set(public_key_inc "${CMAKE_CURRENT_BINARY_DIR}/rtbus_image_public_key.inc")
    generate_inc_file_for_target(app "${RTBUS_IMAGE_PUBLIC_KEY_FILE}" "${public_key_inc}")
    target_include_directories(app PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_sources(app PRIVATE services/rtbus_image_verify.c)
  endif()
endif()
